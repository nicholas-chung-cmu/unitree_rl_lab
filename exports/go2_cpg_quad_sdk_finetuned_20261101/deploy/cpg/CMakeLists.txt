cmake_minimum_required(VERSION 3.16)
project(go2_cpg_wireless LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(unitree_sdk2 REQUIRED)
add_executable(wireless_receiver wireless_receiver.cpp)
target_link_libraries(wireless_receiver PRIVATE unitree_sdk2)
option(CPG_WIRELESS_TESTING "Build isolated wireless-input publisher fixture" OFF)
if(CPG_WIRELESS_TESTING)
  add_executable(cpg_wireless_publisher ../../tests/cpg_wireless_publisher.cpp)
  target_link_libraries(cpg_wireless_publisher PRIVATE unitree_sdk2)
endif()
